BEGIN;

-- 1. TOGGLE LIKE RPC (Atomic)
CREATE OR REPLACE FUNCTION toggle_post_like(p_post_id uuid, p_user_id uuid)
RETURNS boolean
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    v_exists boolean;
BEGIN
    SELECT EXISTS(SELECT 1 FROM public.community_post_likes WHERE post_id = p_post_id AND user_id = p_user_id) INTO v_exists;
    
    IF v_exists THEN
        DELETE FROM public.community_post_likes WHERE post_id = p_post_id AND user_id = p_user_id;
        UPDATE public.community_posts SET likes_count = GREATEST(0, likes_count - 1) WHERE id = p_post_id;
        RETURN false; -- Unliked
    ELSE
        INSERT INTO public.community_post_likes (post_id, user_id) VALUES (p_post_id, p_user_id);
        UPDATE public.community_posts SET likes_count = likes_count + 1 WHERE id = p_post_id;
        RETURN true; -- Liked
    END IF;
END;
$$;

-- 2. TOGGLE PRAYER RPC (Atomic)
CREATE OR REPLACE FUNCTION toggle_post_prayer(p_post_id uuid, p_user_id uuid)
RETURNS boolean
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    v_exists boolean;
BEGIN
    SELECT EXISTS(SELECT 1 FROM public.community_post_prayers WHERE post_id = p_post_id AND user_id = p_user_id) INTO v_exists;
    
    IF v_exists THEN
        DELETE FROM public.community_post_prayers WHERE post_id = p_post_id AND user_id = p_user_id;
        UPDATE public.community_posts SET prayer_count = GREATEST(0, prayer_count - 1) WHERE id = p_post_id;
        RETURN false; -- Unprayed
    ELSE
        INSERT INTO public.community_post_prayers (post_id, user_id) VALUES (p_post_id, p_user_id);
        UPDATE public.community_posts SET prayer_count = prayer_count + 1 WHERE id = p_post_id;
        RETURN true; -- Prayed
    END IF;
END;
$$;

-- 3. COMMENT COUNTER TRIGGER
-- Automatically updates comments_count on the post when a comment is added/removed
CREATE OR REPLACE FUNCTION update_post_comment_count()
RETURNS TRIGGER AS $$
BEGIN
    IF (TG_OP = 'INSERT') THEN
        UPDATE public.community_posts 
        SET comments_count = comments_count + 1,
            -- Optional: Bump post to top on new comment? Uncomment if desired
            -- created_at = now() 
        WHERE id = NEW.post_id;
        RETURN NEW;
    ELSIF (TG_OP = 'DELETE') THEN
        UPDATE public.community_posts 
        SET comments_count = GREATEST(0, comments_count - 1) 
        WHERE id = OLD.post_id;
        RETURN OLD;
    END IF;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Drop existing trigger to avoid conflicts
DROP TRIGGER IF EXISTS tr_update_post_comment_count ON public.community_comments;

CREATE TRIGGER tr_update_post_comment_count
AFTER INSERT OR DELETE ON public.community_comments
FOR EACH ROW EXECUTE FUNCTION update_post_comment_count();

-- 4. ENSURE TABLES EXIST (Safety Check)
CREATE TABLE IF NOT EXISTS public.community_post_likes (
  post_id uuid REFERENCES public.community_posts(id) ON DELETE CASCADE,
  user_id uuid REFERENCES public.users(id),
  created_at timestamp with time zone DEFAULT now(),
  PRIMARY KEY (post_id, user_id)
);

CREATE TABLE IF NOT EXISTS public.community_post_prayers (
  post_id uuid REFERENCES public.community_posts(id) ON DELETE CASCADE,
  user_id uuid REFERENCES public.users(id),
  created_at timestamp with time zone DEFAULT now(),
  PRIMARY KEY (post_id, user_id)
);

-- RLS
ALTER TABLE public.community_post_likes ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.community_post_prayers ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.community_comments ENABLE ROW LEVEL SECURITY;

-- Policies (Drop first to update)
DROP POLICY IF EXISTS "Public Read Likes" ON public.community_post_likes;
CREATE POLICY "Public Read Likes" ON public.community_post_likes FOR SELECT USING (true);

DROP POLICY IF EXISTS "Auth Manage Likes" ON public.community_post_likes;
CREATE POLICY "Auth Manage Likes" ON public.community_post_likes FOR ALL USING (auth.uid() = user_id);

DROP POLICY IF EXISTS "Public Read Prayers" ON public.community_post_prayers;
CREATE POLICY "Public Read Prayers" ON public.community_post_prayers FOR SELECT USING (true);

DROP POLICY IF EXISTS "Auth Manage Prayers" ON public.community_post_prayers;
CREATE POLICY "Auth Manage Prayers" ON public.community_post_prayers FOR ALL USING (auth.uid() = user_id);

DROP POLICY IF EXISTS "Public Read Comments" ON public.community_comments;
CREATE POLICY "Public Read Comments" ON public.community_comments FOR SELECT USING (true);

DROP POLICY IF EXISTS "Auth Manage Comments" ON public.community_comments;
CREATE POLICY "Auth Manage Comments" ON public.community_comments FOR ALL USING (auth.uid() = user_id);

COMMIT;
